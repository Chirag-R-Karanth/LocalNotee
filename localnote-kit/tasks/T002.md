### AppState.kt (create exactly)
```kotlin
package com.localnote.app

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue

/** Top-level screens. */
sealed interface Screen {
    object Library : Screen
    data class Editor(val projectId: String) : Screen
}

/** Holds which screen is showing. */
class AppState {
    var screen: Screen by mutableStateOf(Screen.Library)
        private set

    /** Show the editor for [projectId]. */
    fun openEditor(projectId: String) { screen = Screen.Editor(projectId) }

    /** Show the library. */
    fun backToLibrary() { screen = Screen.Library }
}
```
